cmake_minimum_required(VERSION 3.22)
project(MAIR_JSONCPP_E2E C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT JSONCPP_SOURCE_DIR)
  message(FATAL_ERROR "JSONCPP_SOURCE_DIR is required")
endif()
# One static library. The upstream default also builds an object library from
# the same sources, which would compile and count every check twice.
set(JSONCPP_WITH_TESTS OFF CACHE BOOL "" FORCE)
set(JSONCPP_WITH_POST_BUILD_UNITTEST OFF CACHE BOOL "" FORCE)
set(JSONCPP_WITH_WARNING_AS_ERROR OFF CACHE BOOL "" FORCE)
set(JSONCPP_WITH_PKGCONFIG_SUPPORT OFF CACHE BOOL "" FORCE)
set(JSONCPP_WITH_CMAKE_PACKAGE OFF CACHE BOOL "" FORCE)
set(JSONCPP_WITH_EXAMPLE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_OBJECT_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(${JSONCPP_SOURCE_DIR} jsoncpp)
add_executable(jsoncpp_fuzzer ${JSONCPP_SOURCE_DIR}/src/test_lib_json/fuzz.cpp)
target_link_libraries(jsoncpp_fuzzer PRIVATE jsoncpp_static ${FUZZER_RUNTIME})
add_executable(jsoncpp_oracle ${CMAKE_CURRENT_LIST_DIR}/oracle.cc)
target_link_libraries(jsoncpp_oracle PRIVATE jsoncpp_static)
