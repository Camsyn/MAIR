cmake_minimum_required(VERSION 3.22)
project(MAIR_RE2_E2E C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT RE2_SOURCE_DIR OR NOT ABSL_SOURCE_DIR)
  message(FATAL_ERROR "RE2_SOURCE_DIR and ABSL_SOURCE_DIR are required")
endif()
set(ABSL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ABSL_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
set(RE2_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(RE2_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(${ABSL_SOURCE_DIR} abseil-cpp)
add_subdirectory(${RE2_SOURCE_DIR} re2)
add_executable(re2_fuzzer ${RE2_SOURCE_DIR}/re2/fuzzing/re2_fuzzer.cc)
target_link_libraries(re2_fuzzer PRIVATE re2::re2 ${FUZZER_RUNTIME})
add_executable(re2_oracle ${CMAKE_CURRENT_LIST_DIR}/oracle.cc)
target_link_libraries(re2_oracle PRIVATE re2::re2)
